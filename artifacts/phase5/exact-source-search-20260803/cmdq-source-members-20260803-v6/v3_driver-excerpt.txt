  121	{
  122		struct cmdqFileNodeStruct *pNode;
  123	
  124		CMDQ_VERBOSE("CMDQ driver open fd=%p begin\n", pFile);
  125	
  126		pFile->private_data = kzalloc(sizeof(struct cmdqFileNodeStruct), GFP_KERNEL);
  127		if (pFile->private_data == NULL) {
  128			CMDQ_ERR("Can't allocate memory for CMDQ file node\n");
  129			return -ENOMEM;
  130		}
  131	
  132		pNode = (struct cmdqFileNodeStruct *) pFile->private_data;
  133		pNode->userPID = current->pid;
  134		pNode->userTGID = current->tgid;
  135	
  136		INIT_LIST_HEAD(&(pNode->taskList));
  137		spin_lock_init(&pNode->nodeLock);
  138	
  139		CMDQ_VERBOSE("CMDQ driver open end\n");

  141		return 0;
  142	}

  145	static int cmdq_release(struct inode *pInode, struct file *pFile)
  146	{
  147		struct cmdqFileNodeStruct *pNode;
  148		unsigned long flags;
  149	
  150		CMDQ_VERBOSE("CMDQ driver release fd=%p begin\n", pFile);
  151	
  152		pNode = (struct cmdqFileNodeStruct *) pFile->private_data;
  153	
  154		if (pNode == NULL) {
  155			CMDQ_ERR("CMDQ file node NULL\n");
  156			return -EFAULT;
  157		}
  158	
  159		spin_lock_irqsave(&pNode->nodeLock, flags);
  160	
  161		/* note that we did not release CMDQ tasks */
  162		/* issued by this file node, */
  163		/* since their HW operation may be pending. */
  164	
  165		spin_unlock_irqrestore(&pNode->nodeLock, flags);
  166	
  167		/* scan through tasks that created by this file node and release them */
  168		cmdq_core_release_task_by_file_node((void *)pNode);
  169		cmdqCoreFreeWriteAddressNode((void *)pNode);
  170	
  171		if (pFile->private_data != NULL) {
  172			kfree(pFile->private_data);
  173			pFile->private_data = NULL;
  174		}
  175	
  176		CMDQ_VERBOSE("CMDQ driver release end\n");
  177	
  178		return 0;

  663	static long cmdq_ioctl(struct file *pFile, unsigned int code, unsigned long param)
  664	{
  665		int32_t status = 0;
  666	
  667		switch (code) {
  668		case CMDQ_IOCTL_QUERY_USAGE:
  669			status = cmdq_driver_ioctl_query_usage(pFile, param);
  670			break;
  671		case CMDQ_IOCTL_QUERY_CAP_BITS:
  672			status = cmdq_driver_ioctl_query_cap_bits(param);
  673			break;
  674		case CMDQ_IOCTL_QUERY_DTS:
  675			status = cmdq_driver_ioctl_query_dts(param);
  676			break;
  677		case CMDQ_IOCTL_NOTIFY_ENGINE:
  678			status = cmdq_driver_ioctl_notify_engine(param);
  679			break;
  680		case CMDQ_IOCTL_ASYNC_EXEC:
  681			CMDQ_MSG("ioctl CMDQ_IOCTL_ASYNC_EXEC\n");
  682			status = mdp_ioctl_async_exec(pFile, param);
  683			break;
  684		case CMDQ_IOCTL_ASYNC_WAIT:
  685			CMDQ_MSG("ioctl CMDQ_IOCTL_ASYNC_WAIT\n");
  686			status = mdp_ioctl_async_wait(param);
  687			break;
  688		case CMDQ_IOCTL_ALLOC_READBACK_SLOTS:
  689			CMDQ_MSG("ioctl CMDQ_IOCTL_ALLOC_READBACK_SLOTS\n");
  690			status = mdp_ioctl_alloc_readback_slots(pFile, param);
  691			break;
  692		case CMDQ_IOCTL_FREE_READBACK_SLOTS:
  693			CMDQ_MSG("ioctl CMDQ_IOCTL_FREE_READBACK_SLOTS\n");
  694			status = mdp_ioctl_free_readback_slots(pFile, param);
  695			break;
  696		case CMDQ_IOCTL_READ_READBACK_SLOTS:
  697			CMDQ_MSG("ioctl CMDQ_IOCTL_READ_READBACK_SLOTS\n");
  698			status = mdp_ioctl_read_readback_slots(param);
  699			break;
  700		default:
  701			CMDQ_ERR("unrecognized ioctl 0x%08x\n", code);
  702			return -ENOIOCTLCMD;
  703		}
  704	
  705		return status;
  706	}
